appDirectives.directive("paginate", [ "$timeout", "$rootScope", function($timeout, $rootScope) {
    return {
        restrict: "C",
        replace: true,
        scope: {
            pages: "@paginatePages",
            currentPage: "="
        },
        template: '<ul class="pagination">' + "</ul>",
        controller: function($scope, $element, $attrs) {
            var displayedPages, edges, halfDisplayed;
            halfDisplayed = 1.5;
            displayedPages = 3;
            edges = 2;
            $scope.getInterval = function() {
                return {
                    start: Math.ceil($scope.currentPage > halfDisplayed ? Math.max(Math.min($scope.currentPage - halfDisplayed, $scope.pages - displayedPages), 0) : 0),
                    end: Math.ceil($scope.currentPage > halfDisplayed ? Math.min($scope.currentPage + halfDisplayed, $scope.pages) : Math.min(displayedPages, $scope.pages))
                };
            };
            $scope.selectPage = function(pageIndex) {
                $scope.currentPage = pageIndex * 1;
                $scope.draw();
                $scope.$apply();
            };
            $scope.appendItem = function(pageIndex, opts) {
                var link, options;
                options = void 0;
                link = void 0;
                pageIndex = pageIndex < 0 ? 0 : pageIndex < $scope.pages ? pageIndex : $scope.pages - 1;
                options = $.extend({
                    text: pageIndex + 1,
                    classes: ""
                }, opts || {});
                if (pageIndex === $scope.currentPage && options.classes !== "prev" && options.classes !== "next") {
                    link = $('<li class="active"><a href="javascript:void(0)">' + options.text + ' <span class="sr-only">(current)</span></a></li>');
                } else {
                    link = $('<li><a href="javascript:void(0)" class="page-link">' + options.text + "</a></li>");
                    link.bind("click", function() {
                        $scope.selectPage(pageIndex);
                    });
                }
                if (options.classes) {
                    link.addClass(options.classes);
                }
                $element.append(link);
            };
            $scope.draw = function() {
                var begin, end, i, interval;
                $($element).empty();
                interval = $scope.getInterval();
                i = void 0;
                if (true) {
                    $scope.appendItem($scope.currentPage - 1, {
                        text: "&#171;",
                        classes: "prev"
                    });
                }
                if (interval.start > 0 && edges > 0) {
                    end = Math.min(edges, interval.start);
                    i = 0;
                    while (i < end) {
                        $scope.appendItem(+i);
                        i++;
                    }
                    if (edges < interval.start) {
                        $element.append('<li><span class="ellipse">...</span></li>');
                    }
                }
                i = interval.start;
                while (i < interval.end) {
                    $scope.appendItem(i);
                    i++;
                }
                if (interval.end < $scope.pages && edges > 0) {
                    if ($scope.pages - edges > interval.end) {
                        $element.append('<li><span class="ellipse">...</span></li>');
                    }
                    begin = Math.max($scope.pages - edges, interval.end);
                    i = begin;
                    while (i < $scope.pages) {
                        $scope.appendItem(i);
                        i++;
                    }
                }
                if (true) {
                    $scope.appendItem($scope.currentPage + 1, {
                        text: " &#187;",
                        classes: "next"
                    });
                }
                if ($(window).scrollTop() > 10) {
                    $("body").stop().animate({
                        scrollTop: 1
                    }, 500);
                }
            };
        },
        link: function(scope, element, attrs, controller) {
            scope.$parent.drawPages = scope.draw;
            $timeout(function() {
                scope.draw();
            }, 2e3);
            $rootScope.draw = scope.draw;
            scope.$watch(scope.paginatePages, function() {
                scope.draw();
            });
        }
    };
} ]);

appDirectives.directive("dropselect", [ "$timeout", function($timeout) {
    return function(scope, element, attrs) {
        scope.anosAtivos = scope.$parent.anosAtivos;
        element.on("click", function(event) {
            var $inp, $target, idx, val;
            $target = $(event.currentTarget);
            $inp = $target.find("input");
            val = $inp.val();
            idx = scope.anosAtivos.indexOf(val);
            if (idx > -1) {
                scope.anosAtivos.splice(idx, 1);
                $timeout(function() {
                    return $inp.prop("checked", false);
                }, 0);
            } else {
                scope.anosAtivos.push(val);
                $timeout(function() {
                    return $inp.prop("checked", true);
                }, 0);
            }
            $(event.target).blur();
            scope.$parent.search();
            return false;
        });
    };
} ]);

appDirectives.directive("mic", [ "$timeout", function($timeout) {
    return {
        restrict: "A",
        link: function(scope, element, attrs) {
            scope.final_transcript = "";
            scope.recognizing = false;
            scope.ignore_onend = void 0;
            scope.start_timestamp = void 0;
            element.unbind("click");
            element.bind("click", function(e) {
                return scope.startButton(e);
            });
            scope.startButton = function(event) {
                if (scope.recognizing === true) {
                    scope.recognition.stop();
                    return;
                }
                scope.final_transcript = "";
                scope.recognition.lang = "pt-BR";
                scope.recognition.start();
                scope.ignore_onend = false;
                $("#q").parent().addClass("input--filled");
                $("#q").attr("placeholder", "Fale sua busca");
                $("#q").val("");
                scope.start_timestamp = event.timeStamp;
            };
            if (!("webkitSpeechRecognition" in window)) {} else {
                $("body").addClass("audiable");
                scope.recognition = new webkitSpeechRecognition();
                scope.recognition.continuous = true;
                scope.recognition.interimResults = true;
                scope.recognition.onstart = function() {
                    scope.recognizing = true;
                    $("#mic").addClass("active").append('<div class="ring"></div>');
                    scope.proveFreio();
                };
                scope.proveFreio = function() {
                    clearTimeout(scope.freio);
                    return scope.freio = setTimeout(function() {
                        return scope.finaliza();
                    }, 2500);
                };
                scope.finaliza = function() {
                    scope.recognition.stop();
                };
                scope.recognition.onerror = function(event) {
                    $("#q").attr("placeholder", "");
                    if (event.error === "no-speech") {
                        scope.speechError("Não consigo ouvir seu microfone, ele está habilitado?");
                        element.removeClass("active").find(".ring").remove();
                        scope.ignore_onend = true;
                    }
                    if (event.error === "audio-capture") {
                        element.removeClass("active").find(".ring").remove();
                        scope.speechError("Não consigo ouvir seu microfone, ele está habilitado?");
                        scope.ignore_onend = true;
                    }
                    if (event.error === "not-allowed") {
                        if (event.timeStamp - scope.start_timestamp < 100) {
                            scope.speechError("Seu microfone está bloqueado para este site. <br>Para que a busca por voz funcione ele deve ser liberado em seu navegador");
                        }
                        scope.ignore_onend = true;
                    }
                };
                scope.speechError = function(msg) {
                    scope.erro = {
                        titulo: "Problemas de configuração",
                        texto: msg,
                        ok: true
                    };
                    return scope.showError();
                };
                scope.recognition.onend = function() {
                    scope.recognizing = false;
                    if (scope.ignore_onend) {
                        return;
                    }
                    element.removeClass("active").find(".ring").remove();
                    if (scope.final_transcript) {
                        scope.recognizing = false;
                        scope.queryTerm = scope.final_transcript;
                        scope.search();
                    }
                };
                return scope.recognition.onresult = function(event) {
                    var i;
                    scope.proveFreio();
                    scope.interim_transcript = "";
                    i = event.resultIndex;
                    while (i < event.results.length) {
                        if (event.results[i].isFinal) {
                            scope.final_transcript += event.results[i][0].transcript;
                        } else {
                            scope.interim_transcript += event.results[i][0].transcript;
                        }
                        ++i;
                    }
                    $("#q").val(scope.final_transcript === "" ? scope.interim_transcript : scope.final_transcript);
                };
            }
        }
    };
} ]);

    $scope.preTerm = function(str) {
        var retorno;
        str = str.replace(RegExp(" ?(.+?)\\+(.+) ?"), ' "$1 $2" ').replace(/\+/gi, " ").trim();
        retorno = str;
        if (str.indexOf('"') === -1) {
            if ($scope.fullSearch === true) {
                retorno = '"' + str + '"';
            }
        }
        if ($scope.fullSearch === true) {
            if ((str.match(/\"/g) || []).length % 2 > 0) {
                retorno = str + '"';
            }
        }
        if ($scope.fullSearch === false) {
            retorno = retorno.replace(/\"/g, "");
        }
        return retorno.replace(/\s\s+/g, " ");
    };

    $scope.search = function() {
        $scope.menuMobile = false;
        if ($scope.queryTerm.length < 3 && !$scope.showCalendar) {
            $scope.erro = {
                titulo: "Erro",
                texto: "O campo para busca deve ser preenchido com pelo menos 3 caracteres"
            };
            $scope.showError();
            return false;
        }
        $(".loader").addClass("loading");
        $scope.hideError();
        $http({
            method: "GET",
            url: BASE_URL + "/busca/busca/buscar/" + "query" + "/" + $scope.current + $scope.dateFilters() + ($scope.anosAtivos.length > 0 ? "/y:" + $scope.anosAtivos.join(",") : "") + "/?1=1" + "&q=" + (!$scope.showCalendar ? $scope.preTerm($scope.queryTerm.replace("&", "%26").replace(/(\/)/g, "%2F").replace(":", "")) || "*" : "*") + (TYPE === "materias" ? "&materias=1" : "") + ($scope.showCalendar ? "&calendario=1" : "") + ($scope.subtema ? "&subtheme=" + $scope.subtema : "")
        }).success(function(data, status, headers, config) {
            var d, dataUtil, i, ids, it, m, mes, res, _i, _j, _k, _len, _len1, _len2, _ref, _ref1, _ref2;
            $(".loader").removeClass("loading");
            $("html, body").animate({
                scrollTop: 0
            }, 200);
            if (!$(".menu").hasClass("searched")) {
                $(".menu").addClass("searched");
            }
            if (data.error || !data.hits) {
                if (console && console.log) {
                    console.log("O erro é: " + data.error.reason);
                }
                if (data.mensagem) {
                    $scope.erro = {
                        titulo: "Erro",
                        texto: data.mensagem
                    };
                } else {
                    $scope.erro = {
                        titulo: "Erro",
                        texto: "Houve um problema ao obter os resultados.<br>Se desejar, tente novamente."
                    };
                }
                $scope.showError();
            } else {
                if (data.aggregations === void 0) {
                    data.aggregations = {
                        FileYear: {
                            buckets: []
                        },
                        Edicoes: {
                            buckets: []
                        },
                        TipoEdicao: {
                            buckets: []
                        }
                    };
                    _ref = data.facets.FileYear.terms;
                    for (_i = 0, _len = _ref.length; _i < _len; _i++) {
                        it = _ref[_i];
                        data.aggregations.FileYear.buckets.push({
                            key: it.term,
                            doc_count: it.count
                        });
                    }
                    _ref1 = data.facets.Edicoes.terms;
                    for (_j = 0, _len1 = _ref1.length; _j < _len1; _j++) {
                        it = _ref1[_j];
                        data.aggregations.Edicoes.buckets.push({
                            key: it.term,
                            doc_count: it.count
                        });
                    }
                    _ref2 = data.facets.TipoEdicao.terms;
                    for (_k = 0, _len2 = _ref2.length; _k < _len2; _k++) {
                        it = _ref2[_k];
                        data.aggregations.TipoEdicao.buckets.push({
                            key: it.term,
                            doc_count: it.count
                        });
                    }
                }
                if ($scope.showCalendar) {
                    res = data.aggregations.group_by_data.buckets;
                    for (i in res) {
                        dataUtil = new Date(res[i].key);
                        d = new Date(dataUtil.getTime() + 6e4 * dataUtil.getTimezoneOffset());
                        mes = +d.getMonth() + 1;
                        ids = [];
                        for (m in res[i].group_by_edicao.buckets) {
                            ids.push($scope.addEvent("[icon-view] Baixar Diário", d, $scope.urlCompleto(res[i].group_by_edicao.buckets[m].key)));
                            ids.push($scope.addEvent("[icon-pdf] Visualizar em PDF", d, $scope.urlPdfPagina(res[i].group_by_edicao.buckets[m].key, 1)));
                        }
                        if (res[i].group_by_edicao.buckets.length > 1) {
                            $scope.validateTipoEdicao(d, ids);
                        }
                    }
                    $scope.renderCalender();
                } else {
                    res = data;
                    if (data.aggregations.FileYear.buckets.length > $(".menu-anos .nav > li").length || $(".menu-anos .nav li").length === 0) {
                        $scope.anos = data.aggregations.FileYear.buckets;
                    }
                    $scope.results = data;
                    $scope.pages = res ? Math.ceil(res.hits.total / $scope.pageSize) : 0;
                    $scope.hits = res.hits.hits;
                    if (res.hits.total === 0 && $scope.fullSearch === "1") {
                        $scope.showEmpty = true;
                        $scope.searchAllTerms();
                    } else {
                        $timeout(function() {
                            $scope.drawPages();
                            if ($scope._isNotMobile) {
                                $("#q").focus();
                            }
                            $scope.populateLocation();
                        }, 300);
                    }
                }
            }
        }).error(function(data, status, headers, config) {});
    };

    $scope.getParam = function(param) {
        var h, palavra, subtermo, subtermos, trms, valor;
        subtermo = $location.$$path;
        subtermo = subtermo.substring(subtermo.lastIndexOf("/p"));
        subtermos = subtermo.split("=");
        trms = {};
        h = void 0;
        h = 0;
        while (h < subtermos.length) {
            if (h > 0) {
                palavra = subtermos[h - 1];
                if (palavra.indexOf("/") === 0) {
                    palavra = palavra.substring(1);
                }
                if (palavra.indexOf("&") > 0) {
                    palavra = palavra.substring(palavra.lastIndexOf("&") + 1);
                }
                valor = subtermos[h];
                if (valor.indexOf("&") > -1 && h !== subtermos.length - 1) {
                    valor = valor.substr(0, valor.lastIndexOf("&"));
                }
                trms[palavra] = valor;
            }
            h++;
        }
        return trms[param];
    };
